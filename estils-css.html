Modificar estilos CSS

La propiedad style de un elemento permite cambiar sus estilos CSS desde JavaScript. Por ejemplo element.style.color, element.style.fontSize o element.style.backgroundColor.

Ejemplo: cambiar el estilo de un texto

Este texto cambiará de estilo al pulsar los botones.

Código explicado

// element.style.propiedad cambia un estilo CSS concreto.
// Las propiedades con guion en CSS (font-size) se escriben
// en camelCase en JavaScript (fontSize).

function cambiarColor() {
  document.getElementById("texto").style.color = "red";
}

function cambiarTamano() {
  document.getElementById("texto").style.fontSize = "28px";
}

function cambiarFondo() {
  document.getElementById("texto").style.backgroundColor = "yellow";
}

// Vaciando las propiedades, el texto vuelve a su estilo original
function reiniciar() {
  var t = document.getElementById("texto");
  t.style.color = "";
  t.style.fontSize = "";
  t.style.backgroundColor = "";
}
